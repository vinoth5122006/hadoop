
package org.cloudbus.cloudsim.examples;

import org.cloudbus.cloudsim.*;
import org.cloudbus.cloudsim.core.CloudSim;
import org.cloudbus.cloudsim.provisioners.*;
import java.util.*;

public class ResourceAllocation {

    public static void main(String[] args) throws Exception {

        CloudSim.init(1, Calendar.getInstance(), false);

        // Create 2 hosts
        List<Host> hosts = new ArrayList<Host>();

        for (int i = 0; i < 2; i++) {
            List<Pe> peList = new ArrayList<Pe>();
            peList.add(new Pe(0, new PeProvisionerSimple(4000)));

            hosts.add(new Host(i,
                new RamProvisionerSimple(8192),
                new BwProvisionerSimple(10000),
                1000000, peList,
                new VmSchedulerTimeShared(peList)));
        }

        DatacenterCharacteristics dc = new DatacenterCharacteristics(
            "x86", "Linux", "Xen", hosts,
            10, 3, 0.05, 0.001, 0);

        new Datacenter("DC", dc,
            new VmAllocationPolicySimple(hosts),
            new LinkedList<Storage>(), 0);

        // Create broker
        DatacenterBroker broker = new DatacenterBroker("Broker");
        int id = broker.getId();

        // Create 4 VMs
        List<Vm> vms = new ArrayList<Vm>();

        for (int i = 0; i < 4; i++) {
            vms.add(new Vm(i, id, 1000, 1, 512,
                1000, 10000, "Xen",
                new CloudletSchedulerTimeShared()));
        }

        broker.submitVmList(vms);

        // Create 8 cloudlets
        List<Cloudlet> tasks = new ArrayList<Cloudlet>();

        for (int i = 0; i < 8; i++) {
            Cloudlet c = new Cloudlet(i, 40000, 1, 300, 300,
                new UtilizationModelFull(),
                new UtilizationModelFull(),
                new UtilizationModelFull());

            c.setUserId(id);
            tasks.add(c);
        }

        broker.submitCloudletList(tasks);

        // Run simulation
        CloudSim.startSimulation();

        // Print results
        System.out.println("\nCloudlet ID\tStatus\tVM ID");

        for (Cloudlet c : broker.getCloudletReceivedList()) {
            System.out.println(c.getCloudletId() + "\t\t" +
                (c.getCloudletStatus() == Cloudlet.SUCCESS
                    ? "SUCCESS" : "FAILED") +
                "\t" + c.getVmId());
        }

        CloudSim.stopSimulation();
        System.out.println("Simulation completed.");
    }
}
