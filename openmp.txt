#include <stdio.h>
#include <omp.h>

int main() {
    int a[15] = {12, 45, 23, 67, 89, 34, 56, 78, 11, 90, 32, 54, 76, 21, 10};
    int i, max = a[0], min = a[0], sum = 0;
    int threads;

    omp_set_num_threads(4);

    #pragma omp parallel reduction(max:max) reduction(min:min) reduction(+:sum)
    {
        #pragma omp single
        threads = omp_get_num_threads();

        #pragma omp for
        for (i = 0; i < 15; i++) {
            if (a[i] > max)
                max = a[i];

            if (a[i] < min)
                min = a[i];

            sum += a[i];
        }
    }

    printf("Number of threads = %d\n", threads);
    printf("Maximum = %d\n", max);
    printf("Minimum = %d\n", min);
    printf("Sum = %d\n", sum);

    return 0;
}
